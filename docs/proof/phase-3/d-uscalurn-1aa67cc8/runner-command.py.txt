def save(name,data):
    with (proof/name).open('xb') as f:
        f.write(data if isinstance(data,bytes) else data.encode('utf-8'))
def hashes():
    paths=[Path('game.contract.json'),Path('docs/prd/TICKET-03.md'),Path('docs/prd/TICKET-02.md'),
           *sorted(Path('chonoi').rglob('*.py')),*sorted(Path('tests').glob('*.py'))]
    return {p.as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}

def flatten(suite):
    return [i for sub in suite for i in flatten(sub)] if isinstance(suite,unittest.TestSuite) else [suite.id()]
def execute(label,suite,adapter=None):
    ids=set(flatten(suite))
    stdout,stderr=io.StringIO(),io.StringIO()
    with contextlib.redirect_stdout(stdout),contextlib.redirect_stderr(stderr):
        result=unittest.TextTestRunner(stream=stderr,verbosity=2).run(suite)
    parent=lambda test:getattr(test,'test_case',test).id()
    failures={parent(t) for t,_ in result.failures}
    errors={parent(t) for t,_ in result.errors}
    skips={parent(t) for t,_ in result.skipped}
    methods={i:'ERROR' if i in errors else 'FAIL' if i in failures else 'SKIP' if i in skips else 'PASS' for i in sorted(ids)}
    counts={s:list(methods.values()).count(s) for s in ('PASS','FAIL','ERROR','SKIP')}
    data={'tests_run':result.testsRun,'counts':counts,'methods':methods,'exit_code':0 if result.wasSuccessful() else 1,
          'failure_entries_including_subtests':len(result.failures),'errors':len(result.errors),'adapter':adapter,
          'failures':[{'test':parent(t),'case':t.id(),'traceback':trace} for t,trace in result.failures],
          'execution_errors':[{'test':parent(t),'traceback':trace} for t,trace in result.errors]}
    header=json.dumps({'revision':revision,'branch':branch,'contract_version':contract['contract_version'],
                       'exit_code_equivalent_to_unittest_cli':data['exit_code'],'counts_by_method':counts,
                       'python':sys.executable,'bytecode_disabled':sys.dont_write_bytecode,'adapter':adapter},indent=2)
    save(label+'.txt',header+'\n\n--- STDOUT ---\n'+stdout.getvalue()+'\n--- STDERR ---\n'+stderr.getvalue())
    summaries[label]=data
    print(label+': '+json.dumps(counts),flush=True)
